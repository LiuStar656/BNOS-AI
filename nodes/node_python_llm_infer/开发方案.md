## 一、节点定位

系统**唯一**的 LLM 推理提供者。本地 llama.cpp 和云端 API 都走此节点，Grok 等其他节点不持有任何推理能力。

本节点不做任何工具调用、不做任何决策，纯粹"输入 prompt -> 输出文本"。所有上下文组装、节标记解析、路由决策都在 AAA 节点完成，本节点只负责把 prompt 送进模型并把原始输出返回。

## 二、来源

新建节点，封装 llama.cpp server API + 云端 API 统一适配层。替代原 AAA 项目中内嵌的 `qwen_api` 调用逻辑，将推理能力从业务逻辑中彻底剥离。

## 三、职责

- 接收 AAA 发来的 prompt（`data_type: "prompt"`）
- 根据 `model_type` 配置，调用本地 llama.cpp 或云端 API 进行推理
- 返回节标记文本给 AAA（`data_type: "text"`）
- **不做任何工具调用、不做任何决策**，纯粹输入 prompt -> 输出文本

## 四、端口定义

### 输入端口

| 端口名 | filter (data_type) | 说明 |
|--------|-------------------|------|
| `prompt` | `prompt` | AAA 拼接好的完整 prompt |

### 输出端口

| 端口名 | data_type | 目标 |
|--------|-----------|------|
| `default` | `text` | AAA `llm_response` 端口 |

输出示例：

```json
{"data_type": "text", "content": "【自然回复】\n你好呀！\n\n【心情】\n开心"}
```

## 六、处理逻辑伪代码

遵循 BNOS 节点开发规范：`main.py` 从 stdin 读 JSON、stdout 写 JSON、不依赖任何 BNOS 模块、多端口用 `filter` 路由不用 if-else 分发。

### main.py 伪代码

```python
#!/usr/bin/env python3
"""
节点处理入口。
从 stdin 读取 JSON -> 处理 -> 写入 stdout。
不依赖任何 BNOS 模块。
"""
import sys
import json
import os
import requests


def main():
    try:
        input_data = json.loads(sys.stdin.read())
    except (json.JSONDecodeError, IOError):
        input_data = {}

    prompt_text = input_data.get("content", "") or input_data.get("prompt", "")

    config = load_config()

    # === 业务逻辑：调用 LLM 推理 ===
    result_text = infer(prompt_text, config)

    # 输出节标记文本
    output = {"data_type": "text", "content": result_text}
    sys.stdout.write(json.dumps(output, ensure_ascii=False))
    sys.stdout.flush()


def infer(prompt_text: str, config: dict) -> str:
    """根据 model_type 调用对应后端"""
    model_type = config.get("model_type", "local_llama")
    api_base = config.get("api_base", "http://localhost:8080/v1")
    max_tokens = config.get("max_tokens", 2048)
    temperature = config.get("temperature", 0.7)

    if model_type == "local_llama":
        # 调用本地 llama.cpp server (OpenAI 兼容格式)
        response = requests.post(
            f"{api_base}/chat/completions",
            json={
                "model": "local",
                "messages": [{"role": "user", "content": prompt_text}],
                "max_tokens": max_tokens,
                "temperature": temperature,
                "stream": False
            },
            timeout=120
        )
        response.raise_for_status()
        result_text = response.json()["choices"][0]["message"]["content"]

    elif model_type == "cloud_openai":
        # 调用云端 API (OpenAI 兼容格式)
        api_key = config.get("api_key", "")
        cloud_model = config.get("cloud_model", "qwen-max")
        headers = {"Authorization": f"Bearer {api_key}"}
        response = requests.post(
            f"{api_base}/chat/completions",
            headers=headers,
            json={
                "model": cloud_model,
                "messages": [{"role": "user", "content": prompt_text}],
                "max_tokens": max_tokens,
                "temperature": temperature,
                "stream": False
            },
            timeout=120
        )
        response.raise_for_status()
        result_text = response.json()["choices"][0]["message"]["content"]

    else:
        result_text = ""

    return result_text


def load_config() -> dict:
    """加载 node_config.json 中的参数"""
    config_path = os.path.join(os.path.dirname(__file__), "node_config.json")
    if os.path.exists(config_path):
        with open(config_path, "r", encoding="utf-8") as f:
            return json.load(f)
    return {}


if __name__ == "__main__":
    main()
```

## 七、三后端设计

```
llm_infer 节点
├── http_server  -> llama-server.exe 常驻 HTTP 服务 (localhost:8080/v1) -> Qwen3-1.7B Q4_K_M
├── cli_local    -> llama-cli.exe 子进程调用（零配置，每次加载模型）     -> Qwen3-1.7B Q4_K_M
└── cloud_openai -> 云端 API (OpenAI 兼容格式)                            -> Qwen-Max / GPT-4o
```

通过 `model_type` 参数切换，HTTP 和 CLI 模式调用层接口统一为 **OpenAI 兼容格式**。

| 后端 | `model_type` | 延迟 | 流式 | 适用场景 |
|------|--------------|------|------|----------|
| HTTP Server | `http_server` | 低（模型常驻） | 支持 | 默认，日常对话 |
| CLI Local | `cli_local` | 高（每次加载） | 不支持 | 零配置部署、低频调用 |
| 云端 API | `cloud_openai` | 取决于网络 | 支持 | 高质量推理、联网 |

### 7.1 HTTP Server 模式（默认）

llama-server.exe 作为**子进程**由 llm_infer 节点启动和管理：

```python
import subprocess
import os
import time
import requests

class LlamaServerBackend:
    """llama.cpp HTTP Server 后端 - 模型常驻，低延迟"""

    def __init__(self, config: dict):
        self.model_path = config["model_path"]
        self.port = config.get("llama_port", 8080)
        self.host = "127.0.0.1"
        self.api_base = f"http://{self.host}:{self.port}/v1"
        self.server_process = None
        self._cli_path = self._find_cli_path("llama-server.exe")

    def _find_cli_path(self, exe_name: str) -> str:
        """查找 llama-server.exe，优先节点自带目录"""
        candidates = [
            os.path.join(os.path.dirname(__file__), "llama_cpp_bin", exe_name),
            os.path.join(os.path.dirname(__file__), exe_name),
        ]
        for path in candidates:
            if os.path.exists(path):
                return path
        raise FileNotFoundError(f"未找到 {exe_name}，请放在 llama_cpp_bin/ 目录下")

    def start(self):
        """启动 llama-server 子进程"""
        cmd = [
            self._cli_path,
            "-m", self.model_path,
            "--host", self.host,
            "--port", str(self.port),
            "-c", "4096",           # context length
            "-ngl", "99",           # GPU layers（全部卸载到 GPU）
            "--no-webui",           # 不启动 WebUI
        ]
        self.server_process = subprocess.Popen(
            cmd,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.PIPE,
            creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0,
        )
        # 等待 server 就绪（轮询 /health）
        self._wait_for_ready(timeout=60)

    def _wait_for_ready(self, timeout=60):
        """等待 llama-server 启动完成"""
        start = time.time()
        while time.time() - start < timeout:
            try:
                resp = requests.get(f"http://{self.host}:{self.port}/health", timeout=2)
                if resp.status_code == 200:
                    return
            except:
                pass
            time.sleep(1)
        raise TimeoutError(f"llama-server 启动超时（{timeout}s）")

    def infer(self, prompt: str, max_tokens: int = 2048, temperature: float = 0.7) -> str:
        """调用 OpenAI 兼容 API 推理"""
        resp = requests.post(
            f"{self.api_base}/chat/completions",
            json={
                "model": "local",
                "messages": [{"role": "user", "content": prompt}],
                "max_tokens": max_tokens,
                "temperature": temperature,
                "stream": False,
            },
            timeout=300,
        )
        return resp.json()["choices"][0]["message"]["content"]

    def stop(self):
        """停止 server 进程"""
        if self.server_process:
            self.server_process.terminate()
            self.server_process.wait(timeout=10)
            self.server_process = None
```

### 7.2 CLI Local 模式（零配置备选）

借鉴 `python_node_sqlite` 节点的 CLI 调用方式，每次调用启动 `llama-cli.exe`：

```python
import subprocess
import tempfile
import os

class LlamaCliBackend:
    """llama.cpp CLI 后端 - 零配置，每次调用加载模型"""

    def __init__(self, config: dict):
        self.model_path = config["model_path"]
        self._cli_path = self._find_cli_path("llama-cli.exe")

    def _find_cli_path(self, exe_name: str) -> str:
        candidates = [
            os.path.join(os.path.dirname(__file__), "llama_cpp_bin", exe_name),
            os.path.join(os.path.dirname(__file__), exe_name),
        ]
        for path in candidates:
            if os.path.exists(path):
                return path
        raise FileNotFoundError(f"未找到 {exe_name}")

    def infer(self, prompt: str, max_tokens: int = 2048, temperature: float = 0.7) -> str:
        """通过 llama-cli 子进程推理"""
        # prompt 写入临时文件（避免命令行长度限制）
        fd, prompt_file = tempfile.mkstemp(suffix='.txt', text=True)
        try:
            os.write(fd, prompt.encode('utf-8'))
            os.close(fd)

            cmd = [
                self._cli_path,
                "-m", self.model_path,
                "-f", prompt_file,
                "-n", str(max_tokens),
                "--temp", str(temperature),
                "--top-p", "0.9",
                "--no-display-prompt",
                "-no-cnv",
            ]

            result = subprocess.run(
                cmd,
                capture_output=True,
                timeout=300,
                creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0,
            )

            raw = result.stdout or b""
            try:
                output = raw.decode('utf-8', errors='replace')
            except:
                output = raw.decode('gbk', errors='replace')

            # 清洗 llama-cli 的调试输出
            lines = output.split("\n")
            generated = []
            debug_prefixes = (
                "load_backend:", "build:", "main:", "llama_model_loader:",
                "load:", "load_tensors:", "llama_context:", "llama_kv_cache:",
                "common_init_from_params:", "system_info:", "sampler:",
                "generate:", "llama_perf_", "=== Running", "Press Ctrl+C",
                "Press Return", "error:", "usage:"
            )
            for line in lines:
                stripped = line.strip()
                if stripped and not stripped.startswith(debug_prefixes):
                    generated.append(stripped)

            return "\n".join(generated).strip()

        except subprocess.TimeoutExpired:
            raise TimeoutError("llama-cli 调用超时")
        finally:
            try:
                os.unlink(prompt_file)
            except:
                pass
```

### 7.4 统一入口

```python
def create_backend(config: dict):
    """根据 model_type 创建对应后端"""
    model_type = config.get("model_type", "http_server")

    if model_type == "http_server":
        backend = LlamaServerBackend(config)
        backend.start()  # 启动 llama-server 子进程
        return backend
    elif model_type == "cli_local":
        return LlamaCliBackend(config)
    elif model_type == "cloud_openai":
        return CloudApiBackend(config)
    else:
        raise ValueError(f"未知 model_type: {model_type}")
```

## 八、设计决策

| 决策 | 原因 |
|------|------|
| 统一推理入口 | 本地 llama.cpp 和云端 API 切换无需修改任何其他节点 |
| Grok 不持算力 | Grok 仅执行工具，工具结果需要推理时回传 AAA -> LLM 节点 |
| 方便调试 | 单独查看 AAA 发出的 prompt 和 LLM 返回的节标记文本 |
| 未来扩展 | Live2D 等节点也可能需要 LLM（直接连到此节点） |
| OpenAI 兼容格式 | llama.cpp server 与主流云端 API 均支持，统一适配层零成本 |

## 十、与原项目的差异

| 原版 AAA | 新设计（独立 LLM 节点） |
|----------|------------------------|
| LLM 调用内嵌在 AAA `main.py` | LLM 推理独立为节点，AAA 只组装 prompt |
| 仅支持 qwen_api 单一后端 | 双后端（local_llama / cloud_openai）一键切换 |
| 推理与业务逻辑耦合 | 纯粹输入 prompt -> 输出文本，无业务逻辑 |
| 无法单独调试推理 | 可单独查看 prompt 与 LLM 返回文本 |
| API Key 硬编码或散落配置 | 统一在 `node_config.json` 的 `password` 类型参数中管理 |
